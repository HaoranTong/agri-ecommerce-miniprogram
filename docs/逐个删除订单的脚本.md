\# 获取全部订单 ID（按空格分隔）

$ids = wp wc shop_order list --user=1 --format=ids

if (-not $ids) {

 Write-Host "没有需要删除的订单。"

 return

}

\# 遍历数组逐个删除

$idsArray = $ids -split '\s+'

foreach ($id in $idsArray) {

 if ([string]::IsNullOrWhiteSpace($id)) { continue }

 Write-Host "删除订单: $id"

 wp wc shop_order delete $id --force=true --user=1

}

Write-Host "批量删除完成。"
